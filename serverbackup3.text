// Validate image uploads once at the API boundary so every route receives a
// decodable raster image rather than trusting the browser-supplied MIME type.
async function validateUploadedImage(req, res, next) {
    if (!req.file) return next();

    const supportedMimeTypes = new Set([
        "image/jpeg",
        "image/jpg",
        "image/png",
        "image/webp"
    ]);

    if (!supportedMimeTypes.has(String(req.file.mimetype || "").toLowerCase())) {
        return res.status(415).json({
            success: false,
            error: "Unsupported image type. Please upload a JPG, PNG, or WEBP image."
        });
    }

    try {
        const metadata = await sharp(req.file.buffer).metadata();
        if (!metadata.width || !metadata.height ||
            !["jpeg", "png", "webp"].includes(metadata.format)) {
            return res.status(400).json({
                success: false,
                error: "The uploaded file is not a valid JPG, PNG, or WEBP image."
            });
        }
        return next();
    } catch (error) {
        return res.status(400).json({
            success: false,
            error: "The uploaded file could not be decoded as an image."
        });
    }
}

function shouldFallbackToPollinations(error) {

    const status = error?.response?.status;

    const code = String(error?.code || "");

    const message = String(error?.message || "").toLowerCase();

    // Client-side/network failures.
    if (
        code === "ECONNABORTED" ||
        code === "ETIMEDOUT" ||
        code === "ECONNRESET" ||
        code === "ECONNREFUSED" ||
        code === "ENETUNREACH" ||
        message.includes("timeout") ||
        message.includes("timed out") ||
        message.includes("socket hang up")
    ) {
        return true;
    }

    // If NVIDIA is temporarily unavailable, rate-limited, or its
    // authentication is unavailable, the backup provider can try.
    if (
        status === 401 ||
        status === 403 ||
        status === 408 ||
        status === 410 ||
        status === 429 ||
        status >= 500
    ) {
        return true;
    }

    // No HTTP response generally means the connection itself failed.
    if (!error?.response) {
        return true;
    }

    return false;
}


function extractKimiContent(responseData, isStream = false) {

    if (isStream) {

        let finalContent = "";

        for (
            const line of String(responseData || "").split("\n")
        ) {

            const trimmed = line.trim();

            if (
                !trimmed ||
                !trimmed.startsWith("data:")
            ) {
                continue;
            }

            const data = trimmed.substring(5).trim();

            if (data === "[DONE]") {
                continue;
            }

            try {

                const parsed = JSON.parse(data);

                const choice =
                    parsed?.choices?.[0];

                const delta =
                    choice?.delta || {};

                let piece =
                    delta?.content;

                if (Array.isArray(piece)) {

                    piece = piece
                        .map(item => {

                            if (
                                typeof item === "string"
                            ) {
                                return item;
                            }

                            return (
                                item?.text ||
                                item?.content ||
                                ""
                            );

                        })
                        .join("");
                }

                if (
                    typeof piece === "string" &&
                    piece
                ) {
                    finalContent += piece;
                }

                if (
                    !piece &&
                    typeof choice?.text === "string"
                ) {
                    finalContent += choice.text;
                }

            } catch (error) {

                // Ignore malformed SSE lines.

            }
        }

        return finalContent.trim();
    }

    const messageContent =
        responseData?.choices?.[0]?.message?.content;

    if (Array.isArray(messageContent)) {

        return messageContent
            .map(item => {

                if (typeof item === "string") {
                    return item;
                }

                return (
                    item?.text ||
                    item?.content ||
                    ""
                );

            })
            .join("")
            .trim();
    }

    if (
        typeof messageContent === "string" &&
        messageContent.trim()
    ) {
        return messageContent.trim();
    }

    if (
        typeof responseData?.choices?.[0]?.text === "string"
    ) {
        return responseData.choices[0].text.trim();
    }

    return "";
}


async function callKimiWithFallback({
    imageDataUrl,
    prompt,
    operationName,
    primaryMaxTokens = 4096,
    fallbackMaxTokens = 4096,
    retryPrimaryWithoutStream = false
}) {

    const messages = [
        {
            role: "user",
            content: [
                {
                    type: "text",
                    text: prompt
                },
                {
                    type: "image_url",
                    image_url: {
                        url: imageDataUrl
                    }
                }
            ]
        }
    ];

    let primaryError = null;

    // --------------------------------------------------------
    // PRIMARY: NVIDIA DEEPSEEK V4.1 FLASH
    // --------------------------------------------------------

    try {

        console.log("");
        console.log("----------------------------------------");
        console.log(`VISION MODEL PRIMARY: NVIDIA ${NVIDIA_VISION_MODEL} (${operationName})`);
        console.log("----------------------------------------");

        const response = await axios.post(
            NVIDIA_KIMI_URL,
            {
                model: NVIDIA_VISION_MODEL,
                messages,
                temperature: 0.2,
                max_tokens: primaryMaxTokens,
                stream: false
            },
            {
                headers: {
                    Authorization:
                        `Bearer ${process.env.NVIDIA_API_KEY}`,
                    "Content-Type": "application/json"
                },
                timeout: KIMI_PRIMARY_TIMEOUT_MS
            }
        );

        let finalContent =
            extractKimiContent(
                response.data,
                false
            );

        if (finalContent) {

            console.log(
                `Kimi provider: NVIDIA (${operationName})`
            );

            return {
                content: finalContent,
                provider: "nvidia"
            };
        }

        // Some Kimi image requests can complete the stream without
        // placing the final answer in delta.content. Preserve the
        // existing selected-tooth retry behavior when requested.
        if (retryPrimaryWithoutStream) {

            console.warn(
                `NVIDIA ${NVIDIA_VISION_MODEL} ${operationName} returned no final content. Retrying once without streaming...`
            );

            const retryResponse = await axios.post(
                NVIDIA_KIMI_URL,
                {
                    model: NVIDIA_VISION_MODEL,
                    messages,
                    reasoning_effort: "low",
                    temperature: 0.2,
                    max_tokens: Math.min(
                        primaryMaxTokens,
                        4096
                    ),
                    stream: false
                },
                {
                    headers: {
                        Authorization:
                            `Bearer ${process.env.NVIDIA_API_KEY}`,
                        "Content-Type": "application/json"
                    },
                    timeout: KIMI_PRIMARY_TIMEOUT_MS
                }
            );

            finalContent =
                extractKimiContent(
                    retryResponse.data,
                    false
                );

            if (finalContent) {

                console.log(
                    `Kimi provider: NVIDIA retry (${operationName})`
                );

                return {
                    content: finalContent,
                    provider: "nvidia"
                };
            }
        }

        primaryError = new Error(
            `NVIDIA ${NVIDIA_VISION_MODEL} returned no final content for ${operationName}.`
        );

    } catch (error) {

        primaryError = error;

        console.warn("");
        console.warn(
            `NVIDIA vision model failed for ${operationName}.`
        );
        console.warn(
            "Reason:",
            error?.response?.data ||
            error?.message ||
            error
        );

    }

    // --------------------------------------------------------
    // FALLBACK DECISION
    // --------------------------------------------------------

    if (
        !shouldFallbackToPollinations(primaryError)
    ) {
        throw primaryError;
    }

    console.warn("");
    console.warn(
        `NVIDIA vision model unavailable for ${operationName}.`
    );
    console.warn(
        "Switching automatically to the Pollinations backup model..."
    );

    // --------------------------------------------------------
    // BACKUP: POLLINATIONS KIMI K3
    // --------------------------------------------------------

    try {

        console.log("");
        console.log("----------------------------------------");
        console.log(
            `VISION BACKUP: POLLINATIONS (${operationName})`
        );
        console.log("----------------------------------------");

        const fallbackResponse = await axios.post(
            POLLINATIONS_KIMI_URL,
            {
                model: "moonshotai/kimi-k3",
                messages,
                reasoning_effort: "low",
                temperature: 1,
                max_tokens: fallbackMaxTokens,
                stream: false
            },
            {
                headers: {
                    Authorization:
                        `Bearer ${process.env.POLLINATIONS_API_KEY}`,
                    "Content-Type": "application/json"
                },
                timeout: KIMI_FALLBACK_TIMEOUT_MS
            }
        );

        const fallbackContent =
            extractKimiContent(
                fallbackResponse.data,
                false
            );

        if (!fallbackContent) {

            throw new Error(
                `Pollinations fallback returned no final content for ${operationName}.`
            );
        }

        console.log(
            `Kimi provider: Pollinations backup (${operationName})`
        );

        return {
            content: fallbackContent,
            provider: "pollinations"
        };

    } catch (fallbackError) {

        console.error("");
        console.error(
            `Pollinations fallback also failed for ${operationName}.`
        );
        console.error(
            fallbackError?.response?.data ||
            fallbackError?.message ||
            fallbackError
        );

        // Surface the backup error, but preserve the fact that the
        // primary provider also failed.
        fallbackError.primaryKimiError =
            primaryError?.message || String(primaryError);

        throw fallbackError;
    }
}

// KIMI REPORT PARSER

function parseKimiReport(rawReport) {

    if (!rawReport) {

        throw new Error(

            "Kimi returned an empty report."

        );

    }



    let cleaned =

        String(rawReport)

            .replace(/```json/gi, "")

            .replace(/```/g, "")

            .trim();



    // Direct JSON

    try {

        return JSON.parse(cleaned);

    } catch (error) {

        // Continue to extraction below

    }



    // Extract JSON object

    const start =

        cleaned.indexOf("{");



    const end =

        cleaned.lastIndexOf("}");



    if (

        start !== -1 &&

        end > start

    ) {

        try {

            return JSON.parse(

                cleaned.slice(

                    start,

                    end + 1

                )

            );

        } catch (error) {

            console.error(

                "Kimi JSON parse failed:",

                error.message

            );

        }

    }



    throw new Error(

        "Kimi returned an invalid analysis format."

    );

}

// We DO NOT scan normal observations for words such as

function isVisualizationAllowed(report) {



    // Explicit Kimi decision

    if (

        report?.visualization_ready === false

    ) {

        return false;

    }



    const quality =

        report?.image_quality || {};



    // Teeth explicitly not visible

    if (

        quality.teeth_visible === false

    ) {

        return false;

    }



    // Explicit unusable flag

    if (

        quality.usable_for_visualization === false

    ) {

        return false;

    }



    // Explicit blocking reasons

    if (

        Array.isArray(

            quality.blocking_reasons

        ) &&

        quality.blocking_reasons.length > 0

    ) {

        return false;

    }



    return true;

}

// GET BLOCKING REASONS

// ============================================================



function getBlockingReasons(report) {



    const reasons = [];



    const quality =

        report?.image_quality || {};



    // Kimi-provided blocking reasons

    if (

        Array.isArray(

            quality.blocking_reasons

        )

    ) {

        reasons.push(

            ...quality.blocking_reasons

                .map(String)

                .filter(Boolean)

        );

    }



    // Teeth visibility

    if (

        quality.teeth_visible === false

    ) {

        reasons.push(

            "Teeth are not clearly visible."

        );

    }



    // Explicit unusable flag

    if (

        quality.usable_for_visualization === false &&

        reasons.length === 0

    ) {

        reasons.push(

            "The photograph does not provide enough clear dental detail for visualization."

        );

    }



    // Explicit Kimi false decision

const VALID_FDI_TEETH = new Set([
    "11","12","13","14","15","16","17","18",
    "21","22","23","24","25","26","27","28",
    "31","32","33","34","35","36","37","38",
    "41","42","43","44","45","46","47","48"

function clampNumber(value, min, max) {
    const n = Number(value);
    if (!Number.isFinite(n)) return min;
    return Math.max(min, Math.min(max, n));
}


// Option 1 prototype: derive simple pixel measurements from Kimi's crown bounding box.
// These are bbox proxies, NOT segmented crown measurements or physical millimetres.
function calculateBboxPixelMeasurements(bbox = {}) {
    const width = Number(bbox.width);
    const height = Number(bbox.height);
    if (!Number.isFinite(width) || !Number.isFinite(height) || width <= 0 || height <= 0) {
        return null;
    }
    return {
        method: "kimi_bbox_proxy",
        width_px: Number(width.toFixed(2)),
        height_px: Number(height.toFixed(2)),
        width_to_height_ratio_percent: Number(((width / height) * 100).toFixed(2)),
        units: "pixels",
        calibrated: false,
        represents_actual_crown_boundary: false,
        caveat: "Approximation derived from Kimi's bounding box; may include background or neighboring pixels. Not a crown measurement, clinical measurement, or millimetre estimate."
    };
}

// Validate client-returned target metadata against the actual image. The browser
// sends back a target from /api/tooth-localize, but the backend must not trust
// client-supplied coordinates without checking their bounds.
async function validateSelectedToothTarget(imageBuffer, target, requestedTooth) {
    if (!target || typeof target !== "object" || Array.isArray(target)) {
        throw new Error("Selected tooth location data is missing or invalid.");
    }

    const tooth = String(target.tooth || "").trim();
    if (!VALID_FDI_TEETH.has(tooth) || tooth !== String(requestedTooth)) {
        throw new Error("Selected tooth does not match a valid FDI localization target.");
    }

    const metadata = await sharp(imageBuffer).metadata();
    const imageWidth = Number(metadata.width || 0);
    const imageHeight = Number(metadata.height || 0);
    if (!imageWidth || !imageHeight) {
        throw new Error("Unable to determine uploaded image dimensions.");
    }

    const bbox = target.bbox || {};
    const center = target.center || {};
    const x = Number(bbox.x);
    const y = Number(bbox.y);
    const width = Number(bbox.width);
    const height = Number(bbox.height);
    const centerX = Number(center.x);
    const centerY = Number(center.y);

    if (![x, y, width, height, centerX, centerY].every(Number.isFinite)) {
        throw new Error("Selected tooth coordinates must be valid pixel values.");
    }
    if (width < 1 || height < 1 || x < 0 || y < 0 ||
        x + width > imageWidth || y + height > imageHeight ||
        centerX < 0 || centerY < 0 || centerX >= imageWidth || centerY >= imageHeight) {
        throw new Error("Selected tooth coordinates fall outside the uploaded image.");
    }

    const roundedX = Math.round(x);
    const roundedY = Math.round(y);
    const roundedWidth = Math.round(width);
    const roundedHeight = Math.round(height);
    const roundedCenterX = Math.round(centerX);
    const roundedCenterY = Math.round(centerY);

    if (roundedWidth < 1 || roundedHeight < 1 || roundedX < 0 || roundedY < 0 ||
        roundedX + roundedWidth > imageWidth || roundedY + roundedHeight > imageHeight ||
        roundedCenterX < 0 || roundedCenterY < 0 ||
        roundedCenterX >= imageWidth || roundedCenterY >= imageHeight) {
        throw new Error("Selected tooth coordinates fall outside the uploaded image.");
    }

    const confidenceNumber = Number(target.confidence);
    const confidence = Number.isFinite(confidenceNumber)
        ? clampNumber(confidenceNumber, 0, 1)
        : 0;

    return {
        tooth,
        label: String(target.label || ""),
        visible: target.visible !== false,
        analyzable: target.analyzable === true,
        confidence,
        reason: String(target.reason || ""),
        center: { x: roundedCenterX, y: roundedCenterY },
        bbox: {
            x: roundedX,
            y: roundedY,
            width: roundedWidth,
            height: roundedHeight
        },
        pixel_measurements: calculateBboxPixelMeasurements({
            width: roundedWidth,
            height: roundedHeight
        })
    };
}

function normalizeToothTargets(targets) {
    if (!Array.isArray(targets)) return [];

    const seen = new Set();
    const normalized = [];

    for (const item of targets) {
        const tooth = String(item?.tooth || "").trim();

        if (!VALID_FDI_TEETH.has(tooth)) continue;
        if (seen.has(tooth)) continue;
        if (item?.visible === false) continue;
        if (!item?.bbox) continue;

        const rawBbox = item.bbox || {};
        const rawCenter = item.center || {
            x: Number(rawBbox.x || 0) + Number(rawBbox.width || 0) / 2,
            y: Number(rawBbox.y || 0) + Number(rawBbox.height || 0) / 2
        };

        const x = Math.max(0, Math.round(Number(rawBbox.x || 0)));
        const y = Math.max(0, Math.round(Number(rawBbox.y || 0)));
        const width = Math.max(1, Math.round(Number(rawBbox.width || 1)));
        const height = Math.max(1, Math.round(Number(rawBbox.height || 1)));

        const centerX = Math.max(0, Math.round(Number(rawCenter.x || 0)));
        const centerY = Math.max(0, Math.round(Number(rawCenter.y || 0)));

        const confidence = clampNumber(
            item?.confidence ?? 0,
            0,
            1
        );

        normalized.push({
            tooth,
            label: String(item?.label || ""),
            visible: true,
            analyzable: item?.analyzable !== false,
            confidence,
            reason: String(item?.reason || ""),
            center: {
                x: centerX,
                y: centerY
            },
            bbox: {
                x,
                y,
                width,
                height
            },
            pixel_measurements: calculateBboxPixelMeasurements({ width, height })
        });

        seen.add(tooth);
    }

    return normalized;
}

async function getRefinementImageBuffer(currentImage) {
    const value = String(currentImage || "").trim();
    if (!value) {
        throw new Error("The current visualization is missing. Generate a visualization first.");
    }

    const dataUrlMatch = value.match(/^data:image\/(png|jpe?g|webp);base64,([A-Za-z0-9+/=\r\n]+)$/i);
    if (dataUrlMatch) {
        const encoded = dataUrlMatch[2].replace(/\s/g, "");
        // Reject oversized text payloads before allocating a decoded buffer.
        if (encoded.length > 21 * 1024 * 1024) {
            throw new Error("The current visualization is too large to refine (maximum 15 MB). ");
        }
        const buffer = Buffer.from(encoded, "base64");
        if (!buffer.length) throw new Error("The current visualization data is invalid.");
        if (buffer.length > 15 * 1024 * 1024) {
            throw new Error("The current visualization is too large to refine (maximum 15 MB).");
        }
        const mimeToken = dataUrlMatch[1].toLowerCase();
        const mimeType = mimeToken === "jpg" || mimeToken === "jpeg"
            ? "image/jpeg"
            : `image/${mimeToken}`;
        // Confirm the data URL actually contains a decodable supported image.
        const metadata = await sharp(buffer).metadata();
        if (!metadata.width || !metadata.height || !["jpeg", "png", "webp"].includes(metadata.format)) {
            throw new Error("The current visualization is not a valid PNG, JPEG, or WEBP image.");
        }
        const detectedMimeType = metadata.format === "jpeg"
            ? "image/jpeg"
            : `image/${metadata.format}`;
        return { buffer, mimeType: detectedMimeType };
    }

    let parsedUrl;
    try {
        parsedUrl = new URL(value);
    } catch {
        throw new Error("The current visualization URL is invalid.");
    }

    // Only fetch HTTPS image URLs hosted by Pollinations; do not allow arbitrary
    // URLs from clients to become a server-side request.
    if (parsedUrl.protocol !== "https:" || !/(^|\.)pollinations\.ai$/i.test(parsedUrl.hostname)) {
        throw new Error("The current visualization must be a valid Pollinations HTTPS image URL.");
    }

    const response = await axios.get(value, {
        responseType: "arraybuffer",
        timeout: 60000,
        maxContentLength: 15 * 1024 * 1024,
        maxBodyLength: 15 * 1024 * 1024,
        maxRedirects: 0
    });
    const mimeType = String(response.headers?.["content-type"] || "image/png").split(";")[0].trim();
    if (!/^image\/(png|jpe?g|webp)$/i.test(mimeType)) {
        throw new Error("The current visualization URL did not return a supported image.");
    }
    return { buffer: Buffer.from(response.data), mimeType };
}

function morphologyDilate(mask, width, height, radius = 1) {
    const output = new Uint8Array(mask.length);

    for (let y = 0; y < height; y++) {
        for (let x = 0; x < width; x++) {
            let found = false;

            for (let dy = -radius; dy <= radius && !found; dy++) {
                for (let dx = -radius; dx <= radius; dx++) {
                    const nx = x + dx;
                    const ny = y + dy;

                    if (
                        nx >= 0 && nx < width &&
                        ny >= 0 && ny < height &&
                        mask[ny * width + nx] > 0
                    ) {
                        found = true;
                        break;
                    }
                }
            }

            output[y * width + x] = found ? 255 : 0;
        }
    }

    return output;
}

function morphologyErode(mask, width, height, radius = 1) {
    const output = new Uint8Array(mask.length);

    for (let y = 0; y < height; y++) {
        for (let x = 0; x < width; x++) {
            let allFilled = true;

            for (let dy = -radius; dy <= radius && allFilled; dy++) {
                for (let dx = -radius; dx <= radius; dx++) {
                    const nx = x + dx;
                    const ny = y + dy;

                    if (
                        nx < 0 || nx >= width ||
                        ny < 0 || ny >= height ||
                        mask[ny * width + nx] === 0
                    ) {
                        allFilled = false;
                        break;
                    }
                }
            }

            output[y * width + x] = allFilled ? 255 : 0;
        }
    }

    return output;
}

function morphologyClose(mask, width, height, radius = 1) {
    return morphologyErode(
        morphologyDilate(mask, width, height, radius),
        width,
        height,
        radius
    );
}

function morphologyOpen(mask, width, height, radius = 1) {
    return morphologyDilate(
        morphologyErode(mask, width, height, radius),
        width,
        height,
        radius
    );
}

function keepBestComponent(mask, width, height, targetX, targetY) {
    const visited = new Uint8Array(mask.length);
    let best = null;

    for (let y = 0; y < height; y++) {
        for (let x = 0; x < width; x++) {
            const start = y * width + x;

            if (mask[start] === 0 || visited[start]) continue;

            const queue = [[x, y]];
            const pixels = [];
            visited[start] = 1;

            let sumX = 0;
            let sumY = 0;

            while (queue.length) {
                const [px, py] = queue.pop();

                pixels.push([px, py]);
                sumX += px;
                sumY += py;

                const neighbors = [
                    [px + 1, py],
                    [px - 1, py],
                    [px, py + 1],
                    [px, py - 1]
                ];

                for (const [nx, ny] of neighbors) {
                    if (
                        nx < 0 || nx >= width ||
                        ny < 0 || ny >= height
                    ) continue;

                    const index = ny * width + nx;

                    if (visited[index] || mask[index] === 0) continue;

                    visited[index] = 1;
                    queue.push([nx, ny]);
                }
            }

            if (pixels.length < 3) continue;

            const centerX = sumX / pixels.length;
            const centerY = sumY / pixels.length;

            const distance = Math.sqrt(
                Math.pow(centerX - targetX, 2) +
                Math.pow(centerY - targetY, 2)
            );

            const score =
                distance -
                Math.min(pixels.length / 10, 30);

            if (!best || score < best.score) {
                best = { pixels, score };
            }
        }
    }

    const output = new Uint8Array(mask.length);

    if (!best) return output;

    for (const [x, y] of best.pixels) {
        output[y * width + x] = 255;
    }

    return output;
}

async function createToothReferenceImage(imageBuffer, target) {
    const metadata = await sharp(imageBuffer).metadata();

    const width = Number(metadata.width || 0);
    const height = Number(metadata.height || 0);

    if (!width || !height) {
        throw new Error("Unable to determine image dimensions.");
    }

    const bbox = target.bbox || {};
    const center = target.center || {
        x: Number(bbox.x || 0) + Number(bbox.width || 0) / 2,
        y: Number(bbox.y || 0) + Number(bbox.height || 0) / 2
    };

    const x = Math.max(0, Math.min(width - 1, Math.round(Number(bbox.x || 0))));
    const y = Math.max(0, Math.min(height - 1, Math.round(Number(bbox.y || 0))));
    const boxWidth = Math.max(1, Math.min(width - x, Math.round(Number(bbox.width || 1))));
    const boxHeight = Math.max(1, Math.min(height - y, Math.round(Number(bbox.height || 1))));

    const cx = Math.max(0, Math.min(width - 1, Math.round(Number(center.x || 0))));
    const cy = Math.max(0, Math.min(height - 1, Math.round(Number(center.y || 0))));

    // Keep the annotation outside the tooth as much as possible.
    const labelWidth = 170;
    const labelHeight = 42;

    let labelX = x + boxWidth + 14;
    if (labelX + labelWidth > width - 8) {
        labelX = Math.max(8, x - labelWidth - 14);
    }

    let labelY = Math.max(8, y - labelHeight - 12);
    if (labelY + labelHeight > height - 8) {
        labelY = Math.max(8, Math.min(height - labelHeight - 8, y + boxHeight + 12));
    }

    const lineEndX = labelX < x
        ? labelX + labelWidth
        : labelX;

    const lineEndY = labelY + labelHeight / 2;

    const svg = `
<svg width="${width}" height="${height}" xmlns="http://www.w3.org/2000/svg">
    <rect
        x="${x}"
        y="${y}"
        width="${boxWidth}"
        height="${boxHeight}"
        fill="none"
        stroke="#00e676"
        stroke-width="3"
        stroke-dasharray="7 5"
        rx="4"
    />

    <circle
        cx="${cx}"
        cy="${cy}"
        r="5"
        fill="#00e676"
        stroke="#ffffff"
        stroke-width="2"
    />

    <line
        x1="${cx}"
        y1="${cy}"
        x2="${lineEndX}"
        y2="${lineEndY}"
        stroke="#00e676"
        stroke-width="3"
    />

    <rect
        x="${labelX}"
        y="${labelY}"
        width="${labelWidth}"
        height="${labelHeight}"
        rx="9"
        fill="#101820"
        fill-opacity="0.90"
    />

    <text
        x="${labelX + 12}"
        y="${labelY + 27}"
        font-family="Arial, sans-serif"
        font-size="19"
        font-weight="700"
        fill="#ffffff"
    >FDI ${String(target.tooth)} ✓</text>
</svg>`;

    return sharp(imageBuffer)
        .composite([
            {
                input: Buffer.from(svg),
                top: 0,
                left: 0
            }
        ])
        .jpeg({ quality: 95 })
        .toBuffer();
}

PRESERVE:
- All teeth other than FDI ${target.tooth}; do not change their shape, position, shade, or texture.
- Gingiva, lips, skin, face, lighting, perspective, camera geometry and background/context.
- The selected tooth's natural proportions, rounded corners and enamel texture except where the specified structural correction requires a change. Do not preserve a visible malposition when alignment correction is enabled.
- Do not add braces, crowns, fillings, veneers, or other dental work unless explicitly requested in the tooth-specific assessment.
- Do not create an artificial, oversized, perfectly uniform or plastic-looking tooth.

OUTPUT:
Return only the edited image crop, with no labels, text, boxes, markers, dots, or leader lines. Do not generate a new face or a full smile photograph.
`;
}


async function generateToothVisualization(
    editingImageBuffer,
    mimeType,
    prompt,
    referenceImages = []
) {
    const form = new FormData();

    // First image is the patient's full photograph. Keep it first so the
    // image-editing model treats it as the image to edit; later images are guides.
    form.append(
        "image",
        editingImageBuffer,
        {
            filename: "patient-smile-editing-source.jpg",
            contentType: mimeType || "image/jpeg"
        }
    );

    // Attach only the selected tooth reference(s) after the patient image.
    for (const [index, reference] of referenceImages.entries()) {
        if (!reference || !Buffer.isBuffer(reference.buffer) || !reference.buffer.length) {
            continue;
        }
        form.append("image", reference.buffer, {
            filename: reference.filename || `fdi-reference-${index + 1}.png`,
            contentType: reference.mimeType || "image/png"
        });
    }

    // Explicitly request an output canvas matching the source aspect ratio.
    // GPT Image sizes must be multiples of 16; cap the longest edge at 2560px
    // for a broadly supported output size. The compositor still rejects an
    // actual result whose aspect ratio differs materially from the source.
    const sourceMeta = await sharp(editingImageBuffer).metadata();
    const sourceWidth = Number(sourceMeta.width || 0);
    const sourceHeight = Number(sourceMeta.height || 0);
    if (!sourceWidth || !sourceHeight) {
        throw new Error("Unable to determine source image dimensions before selected-tooth generation.");
    }

    const ratio = sourceWidth / sourceHeight;
    if (ratio < (1 / 3) || ratio > 3) {
        throw new Error("The source photo aspect ratio is outside the image model's supported 1:3 to 3:1 range. Please crop or resize the photo before precise tooth editing.");
    }

    const scale = Math.min(1, 2560 / Math.max(sourceWidth, sourceHeight));
    const roundTo16 = value => Math.max(16, Math.round(value / 16) * 16);
    const outputWidth = roundTo16(sourceWidth * scale);
    const outputHeight = roundTo16(sourceHeight * scale);
    const requestedRatio = outputWidth / outputHeight;
    console.log("Selected-tooth output canvas requested:", {
        source: `${sourceWidth}x${sourceHeight}`,
        requested: `${outputWidth}x${outputHeight}`,
        sourceAspectRatio: Number(ratio.toFixed(5)),
        requestedAspectRatio: Number(requestedRatio.toFixed(5))
    });

    form.append("size", `${outputWidth}x${outputHeight}`);
    form.append("prompt", prompt);
    form.append("model", "openai/gpt-image-2");
    form.append("quality", "high");
    form.append("output_format", "png");

    const response = await axios.post(
        "https://gen.pollinations.ai/v1/images/edits",
        form,
        {
            headers: {
                ...form.getHeaders(),
                Authorization: `Bearer ${process.env.POLLINATIONS_API_KEY}`
            },
            responseType: "json",
            maxContentLength: Infinity,
            maxBodyLength: Infinity,
            timeout: 300000
        }
    );

    return response.data;
}

function dataUrlToBuffer(dataUrl) {
    if (typeof dataUrl !== "string") {
        throw new Error("Generated image is not a valid data URL.");
    }

    const match = dataUrl.match(/^data:[^;]+;base64,(.+)$/s);

    if (!match) {
        throw new Error("Generated image data URL is invalid.");
    }

    return Buffer.from(match[1], "base64");
}


async function generatedImageToBuffer(generatedImage) {
    if (!generatedImage) {
        throw new Error("No generated image was returned by MAI Image 2.6.");
    }

    if (Buffer.isBuffer(generatedImage)) {
        return generatedImage;
    }

    if (generatedImage.startsWith("data:")) {
        return dataUrlToBuffer(generatedImage);
    }

    if (/^https?:\/\//i.test(generatedImage)) {
        const response = await axios.get(generatedImage, {
            responseType: "arraybuffer",
            timeout: 300000,
            maxContentLength: Infinity,
            maxBodyLength: Infinity
        });

        return Buffer.from(response.data);
    }

    throw new Error("Unsupported generated image format.");
}


async function createIsolatedToothEditingCrop(
    imageBuffer,
    target
) {
    const metadata = await sharp(imageBuffer).metadata();
    const imageWidth = Number(metadata.width || 0);
    const imageHeight = Number(metadata.height || 0);

    if (!imageWidth || !imageHeight) {
        throw new Error("Unable to determine original image dimensions.");
    }

    const bbox = target.bbox || {};

    const bx = Number(bbox.x || 0);
    const by = Number(bbox.y || 0);
    const bw = Number(bbox.width || 0);
    const bh = Number(bbox.height || 0);

    if (bw <= 0 || bh <= 0) {
        throw new Error("Selected tooth has an invalid bounding box.");
    }

    // Give MAI useful context around the selected tooth, but do not send
    // the entire photograph for the precise editing operation.
    const padX = Math.max(12, Math.round(bw * 0.45));
    const padY = Math.max(12, Math.round(bh * 0.45));

    const left = Math.max(0, Math.round(bx - padX));
    const top = Math.max(0, Math.round(by - padY));
    const right = Math.min(imageWidth, Math.round(bx + bw + padX));
    const bottom = Math.min(imageHeight, Math.round(by + bh + padY));

    const cropWidth = Math.max(1, right - left);
    const cropHeight = Math.max(1, bottom - top);

    const cropBuffer = await sharp(imageBuffer)
        .extract({
            left,
            top,
            width: cropWidth,
            height: cropHeight
        })
        .jpeg({ quality: 98 })
        .toBuffer();

    return {
        buffer: cropBuffer,
        left,
        top,
        width: cropWidth,
        height: cropHeight,
        toothX: Math.max(0, Math.round(bx - left)),
        toothY: Math.max(0, Math.round(by - top)),
        toothWidth: Math.min(cropWidth, Math.round(bw)),
        toothHeight: Math.min(cropHeight, Math.round(bh))
    };
}


async function validateSelectedToothEdit(originalCropBuffer, editedCropBuffer, target, toothReport) {
    // Compare original (left) and edited (right) crops with a vision-based QA gate.
    // This is a visual quality check, not a clinical diagnosis or guarantee.
    const originalMeta = await sharp(originalCropBuffer).metadata();
    const editedMeta = await sharp(editedCropBuffer).metadata();
    const panelWidth = Math.max(1, Math.min(700, Number(originalMeta.width || 0), Number(editedMeta.width || 0)));
    const panelHeight = Math.max(1, Math.min(700, Number(originalMeta.height || 0), Number(editedMeta.height || 0)));
    const left = await sharp(originalCropBuffer).resize(panelWidth, panelHeight, { fit: "fill" }).png().toBuffer();
    const right = await sharp(editedCropBuffer).resize(panelWidth, panelHeight, { fit: "fill" }).png().toBuffer();
    const comparison = await sharp({
        create: { width: panelWidth * 2, height: panelHeight, channels: 3, background: "white" }
    }).composite([
        { input: left, left: 0, top: 0 },
        { input: right, left: panelWidth, top: 0 }
    ]).png().toBuffer();
    const imageDataUrl = `data:image/png;base64,${comparison.toString("base64")}`;
    const prompt = `STRICT NO-OP REJECTION REVIEW. The comparison image shows ORIGINAL crop on the LEFT and EDITED crop on the RIGHT. Selected tooth: FDI ${target.tooth}. Intended change: ${JSON.stringify({ summary: toothReport?.summary, visualization: toothReport?.visualization, visualization_instructions: toothReport?.visualization_instructions })}. Compare the selected tooth's visible outline/geometry, not just color or texture. Return JSON only with keys pass (boolean), correction_visible (boolean), target_tooth_preserved (boolean), unintended_changes (boolean), invented_defect (boolean), reason (short string). Set correction_visible=false and pass=false if the requested structural change cannot be clearly seen at normal viewing size, if the crown silhouette is effectively unchanged, or if only shade/highlights/texture changed. Pass only if the requested feature has a perceptible visible change, the tooth remains plausible and recognizable, preserved features remain consistent, and no obvious artifacts or neighboring-tooth changes appear. Fail if a structural change was requested but the result is near-identical, or if an unsupported notch/chip/defect was introduced. Do not diagnose disease; assess only visible image-edit quality.`;
    const review = await callKimiWithFallback({
        imageDataUrl,
        prompt,
        operationName: `selected tooth ${target.tooth} edit QA`,
        primaryMaxTokens: 1200,
        fallbackMaxTokens: 1200,
        retryPrimaryWithoutStream: true
    });
    let parsed;
    try {
        const raw = String(review.content || "").replace(/^```(?:json)?\s*/i, "").replace(/\s*```$/, "").trim();
        const match = raw.match(/\{[\s\S]*\}/);
        parsed = JSON.parse(match ? match[0] : raw);
    } catch (error) {
        const qaError = new Error("Selected-tooth visual QA returned invalid JSON; edit was not approved.");
        qaError.code = "TOOTH_EDIT_QA_FAILED";
        throw qaError;
    }
    console.log("Selected-tooth visual QA:", parsed);
    if (parsed.pass !== true || parsed.correction_visible !== true || parsed.target_tooth_preserved !== true || parsed.unintended_changes === true || parsed.invented_defect === true) {
        const qaError = new Error(`Selected-tooth edit failed visual QA: ${String(parsed.reason || "quality checks did not pass").slice(0, 300)}`);
        qaError.code = "TOOTH_EDIT_QA_FAILED";
        qaError.qaResult = parsed;
        throw qaError;
    }
    return parsed;
}

async function compositeIsolatedToothOntoOriginal(
    originalBuffer,
    generatedCropBuffer,
    cropInfo
) {
    // Use a smooth geometric mask around the selected-tooth localization box.
    // Do NOT derive alpha from original enamel pixels: that clips any newly
    // generated crown silhouette to the old tooth outline and creates notches.
    // This is an interim blend mask, not clinical-grade tooth segmentation.
    const width = Math.round(Number(cropInfo.width));
    const height = Math.round(Number(cropInfo.height));
    const left = Math.round(Number(cropInfo.left));
    const top = Math.round(Number(cropInfo.top));
    const toothX = Number(cropInfo.toothX);
    const toothY = Number(cropInfo.toothY);
    const toothWidth = Math.max(1, Number(cropInfo.toothWidth));
    const toothHeight = Math.max(1, Number(cropInfo.toothHeight));

    if (!width || !height || width < 1 || height < 1 ||
        left < 0 || top < 0 ||
        !Number.isFinite(toothX) || !Number.isFinite(toothY) ||
        !Number.isFinite(toothWidth) || !Number.isFinite(toothHeight)) {
        throw new Error("Invalid selected-tooth crop dimensions for compositing.");
    }

    // Expand the target slightly to accommodate the new crown silhouette.
    // Keep the mask bounded to the localized target region to reduce changes
    // to adjacent teeth and surrounding tissue.
    const padX = Math.max(2, Math.round(toothWidth * 0.10));
    const padY = Math.max(2, Math.round(toothHeight * 0.07));
    const x0 = Math.max(0, toothX - padX);
    const y0 = Math.max(0, toothY - padY);
    const x1 = Math.min(width, toothX + toothWidth + padX);
    const y1 = Math.min(height, toothY + toothHeight + padY);
    const maskWidth = Math.max(1, x1 - x0);
    const maskHeight = Math.max(1, y1 - y0);

    // SVG renders a clean, antialiased oval with transparent surroundings.
    // A moderate feather softens the boundary without smearing the crown.
    const rx = Math.max(1, maskWidth * 0.49);
    const ry = Math.max(1, maskHeight * 0.49);
    const feather = Math.max(1.2, Math.min(3.5, Math.min(toothWidth, toothHeight) * 0.025));
    const maskSvg = Buffer.from(`
      <svg width="${width}" height="${height}" viewBox="0 0 ${width} ${height}" xmlns="http://www.w3.org/2000/svg">
        <defs>
          <filter id="feather" x="-20%" y="-20%" width="140%" height="140%">
            <feGaussianBlur stdDeviation="${feather}"/>
          </filter>
        </defs>
        <ellipse
          cx="${(x0 + x1) / 2}"
          cy="${(y0 + y1) / 2}"
          rx="${rx}"
          ry="${ry}"
          fill="white"
          filter="url(#feather)"
        />
      </svg>
    `);

    const maskRaw = await sharp(maskSvg)
        .greyscale()
        .removeAlpha()
        .raw()
        .toBuffer();

    const expectedMaskLength = width * height;
    if (maskRaw.length !== expectedMaskLength) {
        throw new Error("Selected-tooth blend mask dimensions do not match the source crop.");
    }

    const { data: generatedRaw, info: generatedInfo } = await sharp(generatedCropBuffer)
        .resize(width, height, { fit: "fill" })
        .ensureAlpha()
        .raw()
        .toBuffer({ resolveWithObject: true });

    if (generatedInfo.channels !== 4 || generatedRaw.length !== expectedMaskLength * 4) {
        throw new Error("Unable to prepare generated crop for masked compositing.");
    }

    for (let i = 0; i < expectedMaskLength; i++) {
        generatedRaw[i * 4 + 3] = maskRaw[i];
    }

    const maskedGeneratedPng = await sharp(generatedRaw, {
        raw: { width, height, channels: 4 }
    }).png().toBuffer();

    return sharp(originalBuffer)
        .composite([{
            input: maskedGeneratedPng,
            left,
            top,
            blend: "over"
        }])
        .jpeg({ quality: 98 })
        .toBuffer();
}



// Composite a full-frame AI result only inside a soft, target-centered crown region.
// The model may alter other teeth, but those pixels are discarded in favor of the original.
// This is a geometric protection mask based on localization bbox, not a true tooth segmentation.
async function compositeFullFrameWithinSelectedTooth(originalBuffer, generatedBuffer, target) {
    const originalMeta = await sharp(originalBuffer).metadata();
    const generatedMeta = await sharp(generatedBuffer).metadata();
    const width = Number(originalMeta.width || 0);
    const height = Number(originalMeta.height || 0);
    if (!width || !height || !generatedMeta.width || !generatedMeta.height) {
        throw new Error("Unable to determine image dimensions for selected-tooth compositing.");
    }

    const originalAspect = width / height;
    const generatedAspect = Number(generatedMeta.width) / Number(generatedMeta.height);
    const aspectDifference = Math.abs(generatedAspect - originalAspect) / originalAspect;
    console.log("Selected-tooth generated image dimensions:", {
        source: `${width}x${height}`,
        generated: `${generatedMeta.width}x${generatedMeta.height}`,
        sourceAspectRatio: Number(originalAspect.toFixed(5)),
        generatedAspectRatio: Number(generatedAspect.toFixed(5)),
        relativeAspectDifferencePercent: Number((aspectDifference * 100).toFixed(2))
    });
    if (!Number.isFinite(generatedAspect) || aspectDifference > 0.02) {
        throw new Error(`The image-editing model returned a different image composition (source ${width}x${height}; generated ${generatedMeta.width}x${generatedMeta.height}; aspect-ratio difference ${(aspectDifference * 100).toFixed(2)}%). Please retry; the source image was not changed.`);
    }

    const bbox = target?.bbox || {};
    const bx = Number(bbox.x);
    const by = Number(bbox.y);
    const bw = Number(bbox.width);
    const bh = Number(bbox.height);
    if (![bx, by, bw, bh].every(Number.isFinite) || bw < 2 || bh < 2 || bx < 0 || by < 0 || bx + bw > width || by + bh > height) {
        throw new Error("Selected-tooth bounding box is invalid for protected compositing.");
    }

    // Small expansion permits modest crown contour changes while limiting spillover.
    const padX = Math.max(2, Math.round(bw * 0.07));
    const padY = Math.max(2, Math.round(bh * 0.05));
    const x0 = Math.max(0, Math.floor(bx - padX));
    const y0 = Math.max(0, Math.floor(by - padY));
    const x1 = Math.min(width, Math.ceil(bx + bw + padX));
    const y1 = Math.min(height, Math.ceil(by + bh + padY));
    const maskWidth = Math.max(1, x1 - x0);
    const maskHeight = Math.max(1, y1 - y0);
    const radius = Math.max(1, Math.min(maskWidth, maskHeight) * 0.12);
    const feather = Math.max(1, Math.min(2.2, Math.min(maskWidth, maskHeight) * 0.025));

    // Rounded crown-like region with a narrow feather. Avoids a rectangular patch edge.
    const maskSvg = Buffer.from(`<svg width="${width}" height="${height}" viewBox="0 0 ${width} ${height}" xmlns="http://www.w3.org/2000/svg"><defs><filter id="soft" x="-10%" y="-10%" width="120%" height="120%"><feGaussianBlur stdDeviation="${feather}"/></filter></defs><rect width="${width}" height="${height}" fill="black"/><rect x="${x0}" y="${y0}" width="${maskWidth}" height="${maskHeight}" rx="${radius}" ry="${radius}" fill="white" filter="url(#soft)"/></svg>`);
    const mask = await sharp(maskSvg).greyscale().removeAlpha().raw().toBuffer();
    const expected = width * height;
    if (mask.length !== expected) throw new Error("Selected-tooth protection mask dimensions are invalid.");

    const { data: generatedRaw, info } = await sharp(generatedBuffer)
        .resize(width, height, { fit: "fill" })
        .removeAlpha()
        .ensureAlpha()
        .raw()
        .toBuffer({ resolveWithObject: true });
    if (info.channels !== 4 || generatedRaw.length !== expected * 4) {
        throw new Error("Unable to prepare generated image for selected-tooth protection.");
    }
    for (let i = 0; i < expected; i++) generatedRaw[i * 4 + 3] = mask[i];
    const overlay = await sharp(generatedRaw, { raw: { width, height, channels: 4 } }).png().toBuffer();
    return sharp(originalBuffer)
        .composite([{ input: overlay, left: 0, top: 0, blend: "over" }])
        .jpeg({ quality: 98 })
        .toBuffer();
}

async function createSmileReplacementCrop(imageBuffer, targets) {
    const metadata = await sharp(imageBuffer).metadata();
    const imageWidth = Number(metadata.width || 0);
    const imageHeight = Number(metadata.height || 0);
    if (!imageWidth || !imageHeight) throw new Error("Unable to determine original image dimensions.");

    const valid = (Array.isArray(targets) ? targets : []).filter(t => {
        const b = t?.bbox || {};
        return t?.visible !== false && Number(b.width) > 0 && Number(b.height) > 0 &&
            [b.x, b.y, b.width, b.height].every(v => Number.isFinite(Number(v)));
    });
    if (valid.length < 4) throw new Error("At least four visible tooth locations are required for full-tooth-set replacement.");

    const boxes = valid.map(t => ({
        x: Math.max(0, Number(t.bbox.x)), y: Math.max(0, Number(t.bbox.y)),
        right: Math.min(imageWidth, Number(t.bbox.x) + Number(t.bbox.width)),
        bottom: Math.min(imageHeight, Number(t.bbox.y) + Number(t.bbox.height)),
        width: Number(t.bbox.width), height: Number(t.bbox.height), tooth: String(t.tooth || "")
    }));
    const minX = Math.min(...boxes.map(b => b.x));
    const minY = Math.min(...boxes.map(b => b.y));
    const maxX = Math.max(...boxes.map(b => b.right));
    const maxY = Math.max(...boxes.map(b => b.bottom));
    const padX = Math.round((maxX - minX) * 0.14);
    const padY = Math.round((maxY - minY) * 0.22);
    const left = Math.max(0, Math.floor(minX - padX));
    const top = Math.max(0, Math.floor(minY - padY));
    const right = Math.min(imageWidth, Math.ceil(maxX + padX));
    const bottom = Math.min(imageHeight, Math.ceil(maxY + padY));
    const width = right - left, height = bottom - top;
    const buffer = await sharp(imageBuffer).extract({ left, top, width, height }).jpeg({ quality: 98 }).toBuffer();
    return {
        buffer, left, top, width, height,
        toothBoxes: boxes.map(b => ({
            tooth: b.tooth,
            x: Math.max(0, b.x - left), y: Math.max(0, b.y - top),
            width: Math.max(1, Math.min(width, b.right - b.x)),
            height: Math.max(1, Math.min(height, b.bottom - b.y))
        }))
    };
}

async function compositeSmileReplacementOntoOriginal(originalBuffer, generatedCropBuffer, cropInfo) {
    const svgShapes = cropInfo.toothBoxes.map(b => {
        // Slightly expand localization boxes to cover the visible crown outline;
        // overlapping ellipses merge into a single mask for neighboring teeth.
        const cx = b.x + b.width / 2;
        const cy = b.y + b.height / 2;
        const rx = Math.max(2, b.width * 0.60);
        const ry = Math.max(2, b.height * 0.57);
        return `<ellipse cx="${cx}" cy="${cy}" rx="${rx}" ry="${ry}" fill="white"/>`;
    }).join("\n");
    const blur = Math.max(2, Math.min(7, Math.round(Math.min(cropInfo.width, cropInfo.height) * 0.012)));
    const maskSvg = `<svg width="${cropInfo.width}" height="${cropInfo.height}" xmlns="http://www.w3.org/2000/svg"><rect width="100%" height="100%" fill="black"/><g filter="url(#soft)">${svgShapes}</g><defs><filter id="soft" x="-20%" y="-20%" width="140%" height="140%"><feGaussianBlur stdDeviation="${blur}"/></filter></defs></svg>`;
    const mask = await sharp(Buffer.from(maskSvg)).png().toBuffer();
    const generated = await sharp(generatedCropBuffer).resize({ width: cropInfo.width, height: cropInfo.height, fit: "fill" }).removeAlpha().joinChannel(mask).png().toBuffer();
    return sharp(originalBuffer).composite([{ input: generated, left: cropInfo.left, top: cropInfo.top }]).jpeg({ quality: 98 }).toBuffer();
}

        primaryMaxTokens: 1000, fallbackMaxTokens: 1000, retryPrimaryWithoutStream: true
    });
    let parsed;
    try {
        const raw = String(review.content || "").replace(/^```(?:json)?\s*/i, "").replace(/\s*```$/, "").trim();
        const match = raw.match(/\{[\s\S]*\}/);
        parsed = JSON.parse(match ? match[0] : raw);
    } catch {
        const err = new Error("Smile replacement quality review returned invalid JSON.");
        err.code = "SMILE_REPLACEMENT_QA_FAILED";
        throw err;
    }
    console.log("Hybrid smile replacement QA:", parsed);
    if (parsed.pass !== true || parsed.teeth_visibly_regenerated !== true || parsed.mouth_context_preserved !== true || parsed.obvious_artifacts === true) {
        const err = new Error(`Hybrid smile replacement failed visual QA: ${String(parsed.reason || "quality checks did not pass").slice(0, 300)}`);
        err.code = "SMILE_REPLACEMENT_QA_FAILED";
        throw err;
    }
    return parsed;
}

async function localizeTeethWithKimi(imageBuffer, mimeType) {

            "Unable to determine image dimensions."
        );

    const prompt = `
You are performing PRECISE INDIVIDUAL TOOTH LOCALIZATION for a dentist.

Use ONLY the original photograph supplied with this request.
Do not use any previous report, hypothetical selection, marker, or annotation.

IMAGE DIMENSIONS:
width: ${imageWidth}
height: ${imageHeight}

Your job is NOT to analyze the whole smile.
Your job is to identify ONLY teeth whose individual visible crowns can be reasonably localized in this exact image.

FDI NUMBERING:
Upper patient's right: 18 17 16 15 14 13 12 11
Upper patient's left: 21 22 23 24 25 26 27 28
Lower patient's right: 48 47 46 45 44 43 42 41
Lower patient's left: 31 32 33 34 35 36 37 38

IMPORTANT FRONTAL-PHOTO RULE:
For a normal frontal photograph, the patient's right side appears on the viewer's left.

DEFINITIONS:
- visible=true means some portion of that tooth is actually visible.
- analyzable=true means you can reliably identify the FDI tooth AND localize its visible crown sufficiently for a tooth-specific analysis.
- A tooth may be visible but not analyzable.
- Do not invent teeth that cannot be identified.
- Do not automatically return all 32 teeth.
- Do not merge two neighboring teeth into one target.
- Do not use the same center point for two different teeth.
- The center should be inside the visible crown whenever reasonably possible.
- The bbox is a localization guide, NOT a segmentation mask.

RETURN ONLY VALID JSON.

Use exactly:
{
  "image_width": ${imageWidth},
  "image_height": ${imageHeight},

    }

COORDINATES:
- Use exact pixel coordinates relative to the supplied image.
- Do NOT normalize to 0-1000.
- x must be 0..${imageWidth}.
- y must be 0..${imageHeight}.
- bbox.x + bbox.width must not exceed ${imageWidth}.
- bbox.y + bbox.height must not exceed ${imageHeight}.

function parseKimiToothLocalization(rawLocalization) {
    if (!rawLocalization) {
        throw new Error("Kimi returned an empty tooth localization result.");
    }

    let cleaned = String(rawLocalization)
        .replace(/```json/gi, "")
        .replace(/```/g, "")
        .trim();

    let parsed;

    try {
        parsed = JSON.parse(cleaned);
    } catch (error) {
        const firstBrace = cleaned.indexOf("{");
        const lastBrace = cleaned.lastIndexOf("}");

        if (firstBrace === -1 || lastBrace <= firstBrace) {
            throw new Error("Kimi returned invalid tooth localization JSON.");
        }

        try {
            parsed = JSON.parse(
                cleaned.substring(firstBrace, lastBrace + 1)
            );
        } catch (innerError) {
            throw new Error("Kimi returned invalid tooth localization JSON.");
        }
    }

    const teeth = Array.isArray(parsed?.teeth)
        ? parsed.teeth
        : Array.isArray(parsed?.tooth_targets)
            ? parsed.tooth_targets
            : [];

    if (!Array.isArray(teeth)) {
        throw new Error("Kimi localization did not return a teeth array.");
    }

    return {
        image_width: Number(parsed?.image_width || 0),
        image_height: Number(parsed?.image_height || 0),
        teeth
    };
}


function normalizePreciseToothTargets(
    teeth,
    imageWidth,
    imageHeight
) {
    if (!Array.isArray(teeth)) return [];

    const seen = new Set();
    const normalized = [];

    for (const item of teeth) {
        const tooth = String(item?.tooth || "").trim();

        if (!VALID_FDI_TEETH.has(tooth)) continue;
        if (seen.has(tooth)) continue;
        if (item?.visible === false) continue;

        const bbox = item?.bbox || {};
        const center = item?.center || {};

        let x = Number(bbox.x);
        let y = Number(bbox.y);
        let width = Number(bbox.width);
        let height = Number(bbox.height);

        if (![x, y, width, height].every(Number.isFinite)) {
            continue;
        }

        x = Math.round(Math.max(0, Math.min(imageWidth - 1, x)));
        y = Math.round(Math.max(0, Math.min(imageHeight - 1, y)));
        width = Math.round(Math.max(1, Math.min(imageWidth - x, width)));
        height = Math.round(Math.max(1, Math.min(imageHeight - y, height)));

        let centerX = Number(center.x);
        let centerY = Number(center.y);

        if (!Number.isFinite(centerX)) {
            centerX = x + width / 2;
        }

        if (!Number.isFinite(centerY)) {
            centerY = y + height / 2;
        }

        centerX = Math.round(Math.max(0, Math.min(imageWidth - 1, centerX)));
        centerY = Math.round(Math.max(0, Math.min(imageHeight - 1, centerY)));

        const rawConfidence = Number(item?.confidence ?? 0);
        const confidence = Number.isFinite(rawConfidence)
            ? Math.max(0, Math.min(1, rawConfidence))
            : 0;

        normalized.push({
            tooth,
            label: String(item?.label || ""),
            visible: true,
            analyzable: item?.analyzable === true,
            confidence,
            center: {
                x: centerX,
                y: centerY
            },
            bbox: {
                x,
                y,
                width,
                height
            },
            reason: String(item?.reason || "")
        });

        seen.add(tooth);
    }

    return normalized;
}

                    pass: true,

            if (!finalBuffer || !visualQa) {
                throw lastQaError || new Error("Selected-tooth visualization did not pass quality checks.");
            }

            }
            const finalImage = "data:image/jpeg;base64," + finalBuffer.toString("base64");

                tooth: gapClosure ? validatedTargets.map(item => item.tooth).join(", ") : tooth,

// API: CALIBRATED PHYSICAL CROWN MEASUREMENTS
// ============================================================
// IMPORTANT: This endpoint does NOT infer millimetres from a tooth bbox.
// The caller must provide (1) a known-size calibration reference measured
// in the same image plane and (2) crown landmark pixel coordinates measured
// on the original uploaded image. For reliable clinical dimensions, prefer
// a calibrated intraoral scan or a controlled, calibrated image setup.
app.post(
    "/api/tooth-measure",
    upload.single("image"),
    validateUploadedImage,
    async (req, res) => {
        try {
            if (!req.file) {
                return res.status(400).json({ success: false, error: "No image uploaded." });
            }

            const parseObject = (value, fieldName) => {
                if (value && typeof value === "object") return value;
                if (typeof value !== "string" || !value.trim()) {
                    throw new Error(`${fieldName} is required as JSON.`);
                }
                try { return JSON.parse(value); }
                catch (_) { throw new Error(`${fieldName} must contain valid JSON.`); }
            };
            const point = (obj, name) => {
                const p = obj && obj[name];
                const x = Number(p?.x);
                const y = Number(p?.y);
                if (!Number.isFinite(x) || !Number.isFinite(y)) {
                    throw new Error(`Missing or invalid pixel landmark: ${name}.`);
                }
                return { x, y };
            };
            const distance = (a, b) => Math.hypot(b.x - a.x, b.y - a.y);

            const calibration = parseObject(req.body.calibration, "calibration");
            const landmarks = parseObject(req.body.landmarks, "landmarks");
            const knownLengthMm = Number(calibration.knownLengthMm);
            if (!Number.isFinite(knownLengthMm) || knownLengthMm <= 0) {
                return res.status(400).json({
                    success: false,
                    error: "calibration.knownLengthMm must be a positive known physical length in millimetres."
                });
            }

            const referenceStart = point(calibration, "referenceStart");
            const referenceEnd = point(calibration, "referenceEnd");
            const calibrationPixels = distance(referenceStart, referenceEnd);
            if (!Number.isFinite(calibrationPixels) || calibrationPixels < 2) {
                return res.status(400).json({
                    success: false,
                    error: "The calibration reference must span at least 2 pixels. Mark its two endpoints accurately."
                });
            }

            const meta = await sharp(req.file.buffer).metadata();
            const imageWidth = Number(meta.width || 0);
            const imageHeight = Number(meta.height || 0);
            if (!imageWidth || !imageHeight) {
                return res.status(400).json({ success: false, error: "Unable to read uploaded image dimensions." });
            }
            const validatePointInImage = (p, name) => {
                if (p.x < 0 || p.y < 0 || p.x >= imageWidth || p.y >= imageHeight) {
                    throw new Error(`${name} must be inside the original image (${imageWidth} × ${imageHeight} px).`);
                }
            };
            validatePointInImage(referenceStart, "calibration.referenceStart");
            validatePointInImage(referenceEnd, "calibration.referenceEnd");

            const gingival = point(landmarks, "gingivalMarginCenter");
            const incisal = point(landmarks, "incisalEdgeCenter");
            const mesial = point(landmarks, "mesialPoint");
            const distal = point(landmarks, "distalPoint");
            for (const [name, p] of Object.entries({
                gingivalMarginCenter: gingival,
                incisalEdgeCenter: incisal,
                mesialPoint: mesial,
                distalPoint: distal
            })) validatePointInImage(p, `landmarks.${name}`);

            const crownHeightPixels = distance(gingival, incisal);
            const crownWidthPixels = distance(mesial, distal);
            if (crownHeightPixels < 2 || crownWidthPixels < 2) {
                return res.status(400).json({ success: false, error: "Crown landmark points are too close together to measure reliably." });
            }

            const mmPerPixel = knownLengthMm / calibrationPixels;
            const crownHeightMm = crownHeightPixels * mmPerPixel;
            const crownWidthMm = crownWidthPixels * mmPerPixel;
            const tooth = String(req.body.tooth || "").trim();

            // Explicit console output so measurements are visible in the terminal.
            console.log("\n========== SIMOURA CALIBRATED CROWN MEASUREMENT ==========");
            console.log("FDI tooth:", tooth || "not supplied");
            console.log("Image dimensions:", `${imageWidth} × ${imageHeight} px`);
            console.log("Known calibration length:", `${knownLengthMm} mm`);
            console.log("Calibration reference measured:", `${calibrationPixels.toFixed(3)} px`);
            console.log("Scale:", `${mmPerPixel.toFixed(8)} mm/px`);
            console.log("Crown height:", `${crownHeightPixels.toFixed(2)} px = ${crownHeightMm.toFixed(2)} mm`);
            console.log("Crown width:", `${crownWidthPixels.toFixed(2)} px = ${crownWidthMm.toFixed(2)} mm`);
            console.log("Height/width ratio:", (crownHeightPixels / crownWidthPixels).toFixed(4));
            console.log("Method: manual landmarks + single-reference calibration; 2D estimate, not clinically validated.");
            console.log("=========================================================\n");

            return res.json({
                success: true,
                tooth: tooth || null,
                units: { image: "pixels", physical: "mm" },
                measurementMethod: "manual crown landmarks + single-reference image calibration",
                image: { widthPx: imageWidth, heightPx: imageHeight },
                calibration: {
                    knownLengthMm,
                    measuredReferenceLengthPx: Number(calibrationPixels.toFixed(3)),
                    millimetresPerPixel: Number(mmPerPixel.toFixed(8)),
                    referenceStart,
                    referenceEnd,
                    requirement: "Reference and tooth must be at effectively the same depth/plane; perspective and lens distortion can invalidate the scale."
                },
                crownMeasurements: {
                    height: {
                        pixels: Number(crownHeightPixels.toFixed(3)),
                        millimetres: Number(crownHeightMm.toFixed(2)),
                        endpoints: { gingivalMarginCenter: gingival, incisalEdgeCenter: incisal }
                    },
                    width: {
                        pixels: Number(crownWidthPixels.toFixed(3)),
                        millimetres: Number(crownWidthMm.toFixed(2)),
                        endpoints: { mesialPoint: mesial, distalPoint: distal }
                    },
                    heightToWidthRatio: Number((crownHeightPixels / crownWidthPixels).toFixed(4))
                },
                limitations: [
                    "These are 2D projected measurements from manually supplied landmarks, not automatically detected crown boundaries.",
                    "A single scale factor does not correct perspective, lens distortion, tooth curvature, or a calibration object at a different depth.",
                    "Do not use these measurements for clinical decisions unless the acquisition and calibration method has been validated by a dental professional."
                ]
            });
        } catch (error) {
            return res.status(400).json({ success: false, error: error.message || "Unable to calculate calibrated crown measurements." });
        }
    }
);

// API: TOOTH MASK
// ============================================================

app.post(
    "/api/tooth-mask",
    upload.single("image"),
    validateUploadedImage,
    async (req, res) => {
        try {
            if (!req.file) {
                return res.status(400).json({
                    success: false,
                    error: "No image uploaded."
                });
            }

            const tooth = String(
                req.body.tooth || ""
            ).trim();

            if (!tooth) {
                return res.status(400).json({
                    success: false,
                    error: "Tooth number is required."
                });
            }

            let report;

            try {
                report = JSON.parse(
                    req.body.report || "{}"
                );
            } catch (error) {
                return res.status(400).json({
                    success: false,
                    error: "Invalid report JSON."
                });
            }

            const toothTargets = Array.isArray(
                report.tooth_targets
            )
                ? report.tooth_targets
                : [];

            const target = toothTargets.find(
                item => String(item?.tooth) === tooth
            );

            if (!target?.bbox) {
                return res.status(404).json({
                    success: false,
                    error:
                        `Tooth ${tooth} was not found in tooth_targets or has no bbox.`
                });
            }

            const metadata = await sharp(
                req.file.buffer
            ).metadata();

            const imageWidth = metadata.width;
            const imageHeight = metadata.height;

            if (!imageWidth || !imageHeight) {
                return res.status(400).json({
                    success: false,
                    error: "Unable to determine image dimensions."
                });
            }

            const bbox = target.bbox;

            const originalX = Math.max(
                0,
                Math.min(
                    imageWidth - 1,
                    Math.round((Number(bbox.x) / 1000) * imageWidth)
                )
            );

            const originalY = Math.max(
                0,
                Math.min(
                    imageHeight - 1,
                    Math.round((Number(bbox.y) / 1000) * imageHeight)
                )
            );

            const originalWidth = Math.max(
                1,
                Math.min(
                    imageWidth - originalX,
                    Math.round((Number(bbox.width) / 1000) * imageWidth)
                )
            );

            const originalHeight = Math.max(
                1,
                Math.min(
                    imageHeight - originalY,
                    Math.round((Number(bbox.height) / 1000) * imageHeight)
                )
            );

            const paddingX = Math.max(
                5,
                Math.round(originalWidth * 0.25)
            );

            const paddingY = Math.max(
                6,
                Math.round(originalHeight * 0.25)
            );

            const cropLeft = Math.max(
                0,
                originalX - paddingX
            );

            const cropTop = Math.max(
                0,
                originalY - paddingY
            );

            const cropRight = Math.min(
                imageWidth,
                originalX + originalWidth + paddingX
            );

            const cropBottom = Math.min(
                imageHeight,
                originalY + originalHeight + paddingY
            );

            const cropWidth = Math.max(
                1,
                cropRight - cropLeft
            );

            const cropHeight = Math.max(
                1,
                cropBottom - cropTop
            );

            console.log("Tooth segmentation:", {
                tooth,
                imageWidth,
                imageHeight,
                bbox,
                originalPixelBox: {
                    x: originalX,
                    y: originalY,
                    width: originalWidth,
                    height: originalHeight
                },
                cropBox: {
                    x: cropLeft,
                    y: cropTop,
                    width: cropWidth,
                    height: cropHeight
                }
            });

            const {
                data,
                info
            } = await sharp(req.file.buffer)
                .extract({
                    left: cropLeft,
                    top: cropTop,
                    width: cropWidth,
                    height: cropHeight
                })
                .removeAlpha()
                .raw()
                .toBuffer({
                    resolveWithObject: true
                });

            const width = info.width;
            const height = info.height;
            const channels = info.channels;

            let mask = new Uint8Array(
                width * height
            );

            // First-pass tooth segmentation.
            // This is intentionally conservative; it will be tuned
            // after inspecting the actual debug crop.
            for (let y = 0; y < height; y++) {
                for (let x = 0; x < width; x++) {
                    const index =
                        (y * width + x) * channels;

                    const r = data[index];
                    const g = data[index + 1];
                    const b = data[index + 2];

                    const max = Math.max(r, g, b);
                    const min = Math.min(r, g, b);

                    const brightness = max;
                    const saturation = max - min;

                    const isBright =
                        brightness >= 125;

                    const isLowSaturation =
                        saturation <= 95;

                    const isWarmWhite =
                        r >= b - 5;

                    if (
                        isBright &&
                        isLowSaturation &&
                        isWarmWhite
                    ) {
                        mask[y * width + x] = 255;
                    }
                }
            }

            mask = morphologyClose(
                mask,
                width,
                height,
                1
            );

            mask = morphologyOpen(
                mask,
                width,
                height,
                1
            );

            const targetCenterX =
                originalX -
                cropLeft +
                originalWidth / 2;

            const targetCenterY =
                originalY -
                cropTop +
                originalHeight / 2;

            mask = keepBestComponent(
                mask,
                width,
                height,
                targetCenterX,
                targetCenterY
            );

            mask = morphologyClose(
                mask,
                width,
                height,
                1
            );

            const cropMaskBuffer = await sharp(
                Buffer.from(mask),
                {
                    raw: {
                        width,
                        height,
                        channels: 1
                    }
                }
            )
                .png()
                .toBuffer();

            const fullMaskRaw = Buffer.alloc(
                imageWidth * imageHeight
            );

            for (let y = 0; y < height; y++) {
                for (let x = 0; x < width; x++) {
                    const fullX = cropLeft + x;
                    const fullY = cropTop + y;

                    if (
                        fullX >= 0 &&
                        fullX < imageWidth &&
                        fullY >= 0 &&
                        fullY < imageHeight
                    ) {
                        fullMaskRaw[
                            fullY * imageWidth + fullX
                        ] =
                            mask[y * width + x];
                    }
                }
            }

            const fullMaskBuffer = await sharp(
                fullMaskRaw,
                {
                    raw: {
                        width: imageWidth,
                        height: imageHeight,
                        channels: 1
                    }
                }
            )
                .png()
                .toBuffer();

            const overlayRaw = Buffer.alloc(
                imageWidth *
                imageHeight *
                4
            );

            for (let i = 0; i < fullMaskRaw.length; i++) {
                const value = fullMaskRaw[i];
                const index = i * 4;

                overlayRaw[index] = 255;
                overlayRaw[index + 1] = 0;
                overlayRaw[index + 2] = 0;
                overlayRaw[index + 3] =
                    Math.round(value * 0.45);
            }

            const overlayBuffer = await sharp(
                overlayRaw,
                {
                    raw: {
                        width: imageWidth,
                        height: imageHeight,
                        channels: 4
                    }
                }
            )
                .png()
                .toBuffer();

            const previewBuffer = await sharp(
                req.file.buffer
            )
                .composite([
                    {
                        input: overlayBuffer,
                        blend: "over"
                    }
                ])
                .jpeg({
                    quality: 92
                })
                .toBuffer();

            const debugOverlayRaw = Buffer.alloc(
                width * height * 4
            );

            for (let i = 0; i < mask.length; i++) {
                const value = mask[i];
                const index = i * 4;

                debugOverlayRaw[index] = 255;
                debugOverlayRaw[index + 1] = 0;
                debugOverlayRaw[index + 2] = 0;
                debugOverlayRaw[index + 3] =
                    Math.round(value * 0.55);
            }

            const debugOverlayBuffer = await sharp(
                debugOverlayRaw,
                {
                    raw: {
                        width,
                        height,
                        channels: 4
                    }
                }
            )
                .png()
                .toBuffer();

            const debugCropBuffer = await sharp(
                req.file.buffer
            )
                .extract({
                    left: cropLeft,
                    top: cropTop,
                    width: cropWidth,
                    height: cropHeight
                })
                .composite([
                    {
                        input: debugOverlayBuffer,
                        blend: "over"
                    }
                ])
                .jpeg({
                    quality: 95
                })
                .toBuffer();

            return res.json({
                success: true,
                tooth,
                bbox,

                pixelBox: {
                    x: originalX,
                    y: originalY,
                    width: originalWidth,
                    height: originalHeight
                },

                cropBox: {
                    x: cropLeft,
                    y: cropTop,
                    width: cropWidth,
                    height: cropHeight
                },

                mask:
                    "data:image/png;base64," +
                    fullMaskBuffer.toString("base64"),

                preview:
                    "data:image/jpeg;base64," +
                    previewBuffer.toString("base64"),

                debugCrop:
                    "data:image/jpeg;base64," +
                    debugCropBuffer.toString("base64")
            });

        } catch (error) {
            console.error(
                "Tooth mask error:",
                error
            );

            return res.status(500).json({
                success: false,
                error:
                    error.message ||
                    "Failed to create tooth mask."
            });
        }
    }
);

// FDI SHAPE VARIANT PREVIEW API (additive; existing routes unchanged)
// Returns thumbnails as data URLs so the existing public app can show the
// actual reference-library images without exposing the library directory.
// ============================================================

app.get("/api/tooth-shape-variants/:tooth", async (req, res) => {
    const tooth = String(req.params.tooth || "").trim();
    if (!VALID_FDI_TEETH.has(tooth)) {
        return res.status(400).json({ success: false, error: "A valid FDI tooth number is required." });
    }

    try {
        const record = getToothReference(tooth);
        const variants = Array.isArray(record?.shape_variants) ? record.shape_variants : [];
        if (!variants.length) {
            return res.status(404).json({ success: false, code: "NO_SHAPE_VARIANTS", variants: [] });
        }

        const libraryRoot = path.resolve(__dirname, "data", "fdi-tooth-library");
        const output = [];
        for (const variant of variants) {
            const rawImagePath = variant?.image || variant?.image_path || variant?.path || "";
            let imageDataUrl = null;
            if (rawImagePath) {
                const candidates = [
                    path.isAbsolute(rawImagePath) ? rawImagePath : path.resolve(libraryRoot, rawImagePath),
                    path.resolve(libraryRoot, "assets", tooth, "variants", `${variant.id}-front.png`),
                    path.resolve(libraryRoot, "assets", tooth, `${variant.id}-front.png`),
                    path.resolve(libraryRoot, "assets", tooth, `${variant.id}.png`),
                    path.resolve(libraryRoot, "assets", tooth, `${variant.id}.png.png`)
                ];
                for (const candidate of candidates) {
                    const absolute = path.resolve(candidate);
                    if (!(absolute === libraryRoot || absolute.startsWith(libraryRoot + path.sep))) continue;
                    try {
                        const buffer = await fs.promises.readFile(absolute);
                        const metadata = await sharp(buffer).metadata();
                        if (!metadata.width || !metadata.height || !["png", "jpeg", "webp"].includes(metadata.format)) continue;
                        const mime = metadata.format === "jpeg" ? "image/jpeg" : `image/${metadata.format}`;
                        imageDataUrl = `data:${mime};base64,${buffer.toString("base64")}`;
                        break;
                    } catch (_) {
                        // Try the next supported filename/path convention.
                    }
                }
            }
            output.push({
                id: String(variant?.id || ""),
                label: String(variant?.label || variant?.name || variant?.id || "Shape variant"),
                description: String(variant?.description || variant?.notes || ""),
                image: imageDataUrl
            });
        }

        return res.json({ success: true, tooth, variants: output });
    } catch (error) {
        console.error("FDI shape variant preview error:", error);
        return res.status(500).json({ success: false, error: "Unable to load shape variants from the FDI reference library." });
    }
});
